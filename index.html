// ===== 社群總控台 - Google Apps Script ===== // 使用方式： // 1. 打開你的 Google 試算表 // 2. 點「擴充功能」→「Apps Script」 // 3. 把這段程式碼全部貼上去（覆蓋原本的內容） // 4. 點「部署」→「管理部署作業」→ 編輯 → 版本選「新版本」→ 部署 // 5. 網址不變，不用重新複製 const TASK_SHEET = '任務打勾'; const DATA_SHEET = '網站資料'; function getOrCreateSheet(name) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); if (name === TASK_SHEET) { sheet.appendRow(['key', 'value', 'timestamp', 'date', 'brand', 'task_name']); } else { sheet.appendRow(['key', 'value', 'timestamp']); } sheet.setFrozenRows(1); } return sheet; } // 確保任務打勾工作表有新欄位標題 function ensureTaskHeaders(sheet) { const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; if (headers.length < 6 || headers[3] !== 'date') { if (!headers[3]) sheet.getRange(1, 4).setValue('date'); if (!headers[4]) sheet.getRange(1, 5).setValue('brand'); if (!headers[5]) sheet.getRange(1, 6).setValue('task_name'); } } function doGet(e) { const action = e.parameter.action; if (action === 'getTasks') { return getAllTasks(); } if (action === 'getData') { return getAllData(); } if (action === 'getSummary') { return getSummary(); } return ContentService.createTextOutput(JSON.stringify({error: 'unknown action'})) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { const data = JSON.parse(e.postData.contents); if (data.action === 'toggle') { return toggleTask(data.key, data.value, data.label, data.brand, data.day); } if (data.action === 'batchUpdate') { return batchUpdate(data.tasks); } if (data.action === 'saveData') { return saveData(data.key, data.value); } if (data.action === 'saveAllData') { return saveAllData(data.items); } if (data.action === 'createCalendarEvent') { return createCalendarEvent(data); } if (data.action === 'deleteCalendarEvent') { return deleteCalendarEvent(data); } return jsonResponse({error: 'unknown action'}); } catch (err) { return jsonResponse({error: err.message}); } } function toggleTask(key, value, label, brand, day) { const sheet = getOrCreateSheet(TASK_SHEET); ensureTaskHeaders(sheet); const data = sheet.getDataRange().getValues(); const dateStr = day ? '4/' + day : ''; let found = false; for (let i = 1; i < data.length; i++) { if (data[i][0] === key) { sheet.getRange(i + 1, 2).setValue(value); sheet.getRange(i + 1, 3).setValue(new Date().toISOString()); // 補上可讀欄位（如果之前沒有的話） if (label && !data[i][5]) sheet.getRange(i + 1, 6).setValue(label); if (brand && !data[i][4]) sheet.getRange(i + 1, 5).setValue(brand); if (dateStr && !data[i][3]) sheet.getRange(i + 1, 4).setValue(dateStr); found = true; break; } } if (!found) { sheet.appendRow([key, value, new Date().toISOString(), dateStr, brand || '', label || '']); } return jsonResponse({ok: true, key: key, value: value}); } function batchUpdate(tasks) { const sheet = getOrCreateSheet(TASK_SHEET); const data = sheet.getDataRange().getValues(); const existing = {}; for (let i = 1; i < data.length; i++) { existing[data[i][0]] = i + 1; } for (const t of tasks) { if (existing[t.key]) { sheet.getRange(existing[t.key], 2).setValue(t.value); sheet.getRange(existing[t.key], 3).setValue(new Date().toISOString()); } else { sheet.appendRow([t.key, t.value, new Date().toISOString()]); } } return jsonResponse({ok: true, count: tasks.length}); } function saveData(key, value) { const sheet = getOrCreateSheet(DATA_SHEET); const data = sheet.getDataRange().getValues(); let found = false; const jsonVal = typeof value === 'string' ? value : JSON.stringify(value); for (let i = 1; i < data.length; i++) { if (data[i][0] === key) { sheet.getRange(i + 1, 2).setValue(jsonVal); sheet.getRange(i + 1, 3).setValue(new Date().toISOString()); found = true; break; } } if (!found) { sheet.appendRow([key, jsonVal, new Date().toISOString()]); } return jsonResponse({ok: true, key: key}); } function saveAllData(items) { const sheet = getOrCreateSheet(DATA_SHEET); const data = sheet.getDataRange().getValues(); const existing = {}; for (let i = 1; i < data.length; i++) { existing[data[i][0]] = i + 1; } for (const item of items) { const jsonVal = typeof item.value === 'string' ? item.value : JSON.stringify(item.value); if (existing[item.key]) { sheet.getRange(existing[item.key], 2).setValue(jsonVal); sheet.getRange(existing[item.key], 3).setValue(new Date().toISOString()); } else { sheet.appendRow([item.key, jsonVal, new Date().toISOString()]); } } return jsonResponse({ok: true, count: items.length}); } function getAllTasks() { const sheet = getOrCreateSheet(TASK_SHEET); const data = sheet.getDataRange().getValues(); const tasks = {}; for (let i = 1; i < data.length; i++) { if (data[i][0]) { tasks[data[i][0]] = data[i][1] === true || data[i][1] === 'TRUE' || data[i][1] === true; } } return jsonResponse(tasks); } function getAllData() { const sheet = getOrCreateSheet(DATA_SHEET); const data = sheet.getDataRange().getValues(); const result = {}; for (let i = 1; i < data.length; i++) { if (data[i][0]) { try { result[data[i][0]] = JSON.parse(data[i][1]); } catch(e) { result[data[i][0]] = data[i][1]; } } } return jsonResponse(result); } function getSummary() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const mainSheet = ss.getSheetByName('4月每日總表'); const mainData = mainSheet ? mainSheet.getDataRange().getValues() : []; let zyPosted = 0, zyWritten = 0, personalPosted = 0, personalWritten = 0; const missedDays = []; for (let i = 2; i < mainData.length; i++) { const row = mainData[i]; if (!row[0] || row[0] === '完成統計') continue; const day = parseInt(String(row[0]).replace('4/', '')); if (isNaN(day)) continue; const zyStatus = row[3] || '待寫'; const pStatus = row[5] || '待寫'; if (zyStatus === '已發') zyPosted++; else if (zyStatus === '已寫') zyWritten++; if (pStatus === '已發') personalPosted++; else if (pStatus === '已寫') personalWritten++; const today = new Date(); const checkDate = new Date(2026, 3, day); if (checkDate < today && zyStatus === '待寫') { missedDays.push({day: day, brand: '智堯', status: zyStatus}); } if (checkDate < today && pStatus === '待寫') { missedDays.push({day: day, brand: '青青', status: pStatus}); } } const taskSheet = getOrCreateSheet(TASK_SHEET); const taskData = taskSheet.getDataRange().getValues(); let tasksCompleted = 0, tasksTotal = 0; for (let i = 1; i < taskData.length; i++) { tasksTotal++; if (taskData[i][1] === true || taskData[i][1] === 'TRUE') tasksCompleted++; } return jsonResponse({ zy: {posted: zyPosted, written: zyWritten, total: 30}, personal: {posted: personalPosted, written: personalWritten, total: 30}, tasks: {completed: tasksCompleted, total: tasksTotal}, missed: missedDays }); } // ===== Google Calendar 連動 ===== function createCalendarEvent(data) { const cal = CalendarApp.getDefaultCalendar(); const year = data.year || 2026; const month = data.month || 4; const day = data.day || 1; const title = data.title || '待辦事項'; const brand = data.brand || ''; const brandNames = {star: '【初星群】', zy: '【智堯】', personal: '【青青】', life: '【生活】'}; const prefix = brandNames[brand] || ''; const fullTitle = prefix + title; const eventDate = new Date(year, month - 1, day); if (data.hour && data.minute !== undefined) { const start = new Date(year, month - 1, day, data.hour, data.minute); const end = new Date(start.getTime() + 60 * 60 * 1000); const event = cal.createEvent(fullTitle, start, end); return jsonResponse({ok: true, eventId: event.getId(), type: 'timed'}); } else { const event = cal.createAllDayEvent(fullTitle, eventDate); return jsonResponse({ok: true, eventId: event.getId(), type: 'allDay'}); } } function deleteCalendarEvent(data) { try { const cal = CalendarApp.getDefaultCalendar(); const event = cal.getEventById(data.eventId); if (event) { event.deleteEvent(); return jsonResponse({ok: true, deleted: true}); } return jsonResponse({ok: false, error: 'event not found'}); } catch(e) { return jsonResponse({ok: false, error: e.message}); } } function jsonResponse(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
